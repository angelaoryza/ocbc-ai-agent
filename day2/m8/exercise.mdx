## The checks

The block at the bottom of the file tests your work. Do not edit it: each `✅` line means one part is done.

```python m8_exercise.py (checks)
# ── Check your work (do not edit) ──────────────────────────────────────────────

if __name__ == "__main__":
    assert hasattr(get_watchlist, "invoke"), "Add @tool above get_watchlist"
    assert get_watchlist.invoke({}) == WATCHLIST, "get_watchlist should return WATCHLIST"
    print("✅ get_watchlist:", get_watchlist.invoke({}))

    result = asyncio.run(get_mcp_answer("How did NISP close on 23 October 2026?"))
    assert result is not None, "get_mcp_answer should return the result of agent.ainvoke"
    used = tools_used(result)
    assert used and all(name in SELECTED for name in used), f"Only SELECTED MCP tools are needed here, the agent used {used}"
    assert "1,185" in result["messages"][-1].text or "1185" in result["messages"][-1].text, "NISP closed at 1,185"
    print("✅ MCP only:", used)

    result = asyncio.run(get_mcp_answer("Which of my watchlist banks closed highest on its last trading day?"))
    used = tools_used(result)
    assert "get_watchlist" in used, f"The agent should look up the watchlist first, it used {used}"
    assert any(name in SELECTED for name in used), f"The agent should then use an MCP tool, it used {used}"
    print("✅ Mixed:", used)
    print("✅ Answer:", result["messages"][-1].text[:150].replace("\n", " "))
    print("\n🎉 Module 8 exercise done!")
```
